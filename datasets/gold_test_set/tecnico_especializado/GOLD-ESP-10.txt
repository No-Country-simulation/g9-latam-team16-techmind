import torch\nimport torch.nn as nn\nimport torch.nn.functional as F\n\nclass DeepTextClassificationCNN(nn.Module):\n    def __init__(self, vocab_size: int, embed_dim: int, num_classes: int, num_filters: int = 128):\n        super(DeepTextClassificationCNN, self).__init__()\n        \n        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)\n        \n        self.conv1 = nn.Conv1d(in_channels=embed_dim, out_channels=num_filters, kernel_size=3, padding=1)\n        self.conv2 = nn.Conv1d(in_channels=embed_dim, out_channels=num_filters, kernel_size=4, padding=2)\n        self.conv3 = nn.Conv1d(in_channels=embed_dim, out_channels=num_filters, kernel_size=5, padding=2)\n        \n        self.dropout = nn.Dropout(p=0.5)\n        self.fc1 = nn.Linear(num_filters * 3, 256)\n        self.fc2 = nn.Linear(256, num_classes)\n\n    def forward(self, text_input):\n        embedded = self.embedding(text_input).permute(0, 2, 1)\n        \n        x1 = F.relu(self.conv1(embedded)).max(dim=2)[0]\n        x2 = F.relu(self.conv2(embedded)).max(dim=2)[0]\n        x3 = F.relu(self.conv3(embedded)).max(dim=2)[0]\n        \n        concatenated = torch.cat((x1, x2, x3), dim=1)\n        dropped = self.dropout(concatenated)\n        dense1 = F.relu(self.fc1(dropped))\n        logits = self.fc2(dense1)\n        \n        return logits En el marco de la modernización de la plataforma, este componente fue refactorizado para garantizar un rendimiento óptimo bajo cargas de trabajo pesadas.
