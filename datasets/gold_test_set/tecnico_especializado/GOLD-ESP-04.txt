WITH MonthlyDocumentStats AS (\n    SELECT \n        d.category,\n        d.author_name,\n        COUNT(d.document_id) AS total_documents,\n        SUM(d.word_count) AS total_words,\n        AVG(d.quality_score) AS avg_quality_score,\n        DENSE_RANK() OVER (\n            PARTITION BY d.category \n            ORDER BY COUNT(d.document_id) DESC, AVG(d.quality_score) DESC\n        ) AS rank_within_category\n    FROM master_documents d\n    WHERE d.created_date >= NOW() - INTERVAL '12 months'\n      AND d.status = 'Approved'\n    GROUP BY d.category, d.author_name\n)\nSELECT \n    category,\n    author_name,\n    total_documents,\n    total_words,\n    ROUND(avg_quality_score::numeric, 2) AS quality_score\nFROM MonthlyDocumentStats\nWHERE rank_within_category <= 3\nORDER BY category ASC, rank_within_category ASC; En el marco de la modernización de la plataforma, este componente fue refactorizado para garantizar un rendimiento óptimo bajo cargas de trabajo pesadas.
