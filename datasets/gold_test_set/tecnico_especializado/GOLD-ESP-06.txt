import { useState, useEffect } from 'react';\n\ninterface FetchOptions {\n  headers?: Record<string, string>;\n  method?: 'GET' | 'POST' | 'PUT' | 'DELETE';\n  body?: any;\n}\n\ninterface FetchResponse<T> {\n  data: T | null;\n  loading: boolean;\n  error: string | null;\n}\n\nexport function useApiFetch<T>(url: string, options?: FetchOptions): FetchResponse<T> {\n  const [data, setData] = useState<T | null>(null);\n  const [loading, setLoading] = useState<boolean>(true);\n  const [error, setError] = useState<string | null>(null);\n\n  useEffect(() => {\n    const controller = new AbortController();\n    const { signal } = controller;\n\n    async function executeFetch() {\n      setLoading(true);\n      try {\n        const response = await fetch(url, {\n          method: options?.method || 'GET',\n          headers: { 'Content-Type': 'application/json', ...options?.headers },\n          body: options?.body ? JSON.stringify(options.body) : undefined,\n          signal,\n        });\n\n        if (!response.ok) {\n          throw new Error(`HTTP Error ${response.status}: ${response.statusText}`);\n        }\n\n        const result: T = await response.json();\n        setData(result);\n        setError(null);\n      } catch (err: any) {\n        if (err.name !== 'AbortError') {\n          setError(err.message || 'Ocurrió un error inesperado');\n        }\n      } finally {\n        setLoading(false);\n      }\n    }\n\n    executeFetch();\n\n    return () => {\n      controller.abort();\n    };\n  }, [url]);\n\n  return { data, loading, error };\n} En el marco de la modernización de la plataforma, este componente fue refactorizado para garantizar un rendimiento óptimo bajo cargas de trabajo pesadas.
